# UFW status
sudo ufw status

# Check active PORTS
sudo lsof -i:80
sudo ss -tlnpu  // all port
sudo kill -9 id

# Port open
sudo ufw allow 80/tcp       # HTTP
sudo ufw allow 443/tcp      # HTTPS
sudo ufw allow 8080         # TCP + UDP

# Port deny
sudo ufw deny 8080/tcp

# regel delete
sudo ufw delete allow 8080/tcp

# UFW change
sudo ufw enable
sudo ufw disable
