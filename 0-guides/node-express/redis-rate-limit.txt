import {client} from "../config/connectRedis.js";

function rateLimiter(limit = 10, window = 60, type = 'ip') {
  return async (req, res, next) => {
    const identifier = {
      ip:     req.ip,
      user:    req.user?.id,
      apiKey: req.headers['x-api-key'],
      global: 'global',
    }[type];

    const key = `rl:${req.path}:${identifier}`;
    const count = await client.incr(key);
    if (count && parseInt(count) === 1) await client.expire(key, window);
    if (count && parseInt(count) > limit) return res.status(429).json({ error: 'Too many requests' });
    next();
  };
}

export default rateLimiter


in router
router.post('/login',    rateLimiter(5,   60, 'ip'));
router.get('/products',  rateLimiter(30,  60, 'user'));
router.get('/report',    rateLimiter(10,  60, 'apiKey'));
app.use(rateLimiter(100, 60, 'global'));
