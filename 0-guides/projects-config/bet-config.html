3. Mongoose Production Konfigürasyonu

const mongooseOptions = {
  maxPoolSize: 100,             // PM2 Cluster başına 100 bağlantı (Toplamda 800 şerit)
  minPoolSize: 10,              // Her an hazırda bekleyecek minimum bağlantı sayısı
  connectTimeoutMS: 10000,      // İlk bağlantı için 10 saniye zaman aşımı
  socketTimeoutMS: 5000,        // Yanıt vermeyen sorgular için 5 saniye zaman aşımı
  keepAlive: true,              // Bağlantıyı işletim sistemi seviyesinde canlı tut
  keepAliveInitialDelay: 300000,// 5 dakikada bir arka planda ping at
  w: 'majority',                // Yazma güvencesi (Veri hem ana hem yedeğe yazılmalı)
  wtimeoutMS: 2000,             // Replikasyon onayı için maksimum 2 saniye bekle
  readPreference: 'secondaryPreferred' // Okuma isteklerini otomatik olarak yedek sunucuya yönlendir
};
