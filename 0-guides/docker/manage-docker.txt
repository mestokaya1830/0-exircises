👉️CREATE IMAGE AND CONTAINER
docker build -t app-name . //imagename
docker run --publish 3000:3000 imagename

👉️REMOVE IMAGE AND CONTAINER
docker ps -a
docker ps
docker kill containerid
docker rm containerid

docker container prune remove //only stopped container
docker rm -f $(docker ps -aq) //remove all containers
docker system prune -a -f //clear all conatiner images networkd and cache

docker images
docker rmi -f imagename
